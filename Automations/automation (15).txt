    value_template: >

    value_template: "{{ states('sensor.r5_charging_remaining_time_api') | int(0) <= 0 }}"

    value_template: "{{ states('<ev-charger-power>') | float(0) <= 0.2 }}"

    data:
      value: "{{ test_start_power }}"

      duration_min: 5

        {{ ((test_end_energy - test_start_energy) / (duration_min / 60)) |
